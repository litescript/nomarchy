# Mod+M runs common/scripts/spotify-limited, which wraps spotify-LAUNCHER, not `spotify`.
# Arch ships the launcher in [extra] and it fetches the real binary into
# ~/.local/share/spotify-launcher/ on first run, so the AUR `spotify` package is not wanted
# here and would be a second copy. The launched processes still report as `spotify`, which
# is what the script's already-running guard and the Mod+M window match both key on.
spotify-launcher

# Docker, ROOTLESS, on every desktop host: lazydocker above is bound to Mod+D everywhere,
# and until 2026-10-04 only caesar had a daemon for it to talk to. There is no rootful
# socket (docker.service stays disabled) and the ufw-docker workaround is deliberately not
# carried, because rootless Docker does not bypass the firewall -- see
# hosts/caesar/ufw/apply-rules. The daemon is docker-rootless-extras (./aur.txt), started
# by its docker.socket user unit (common/bootstrap/units); the client is pointed at it by
# the `rootless docker context` check in common/bootstrap/user-checks.
docker

# umbriel's and its portal's BUILD dependencies (their PKGBUILDs' makedepends, less git and
# pkgconf, which base has). The installer builds them with makepkg -s, but umbriel-update
# rebuilds WITHOUT -s -- it builds, gates the new binary on the live config, then installs
# -- so these must stay installed. Left as makepkg's leftovers they were orphans (meson,
# nlohmann-json in the census) that one `pacman -Rns $(pacman -Qdtq)` would remove, and
# the next umbriel-update would fail to build. Declared, they are owned; and the installer's
# makepkg --rmdeps never removes what it did not install.
mesa
meson
ninja
nlohmann-json
tomlplusplus
wayland-protocols
